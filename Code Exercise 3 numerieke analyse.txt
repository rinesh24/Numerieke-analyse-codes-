
x = [0; 0; 0; 0];


tol = 1e-6;


maxiter = 1000;


for k = 1:maxiter

    x_old = x;

    % theta 1
    x(1) = (120 + 4*x(2) + 3*x(3) + 2*x(4))/18;

    % theta 2
    x(2) = (180 + 4*x(1) + 5*x(3) + 3*x(4))/22;

    % theta 3
    x(3) = (90 + 3*x(1) + 5*x(2) + 4*x(4))/20;

    % theta 4
    x(4) = (150 + 2*x(1) + 3*x(2) + 4*x(3))/24;

    
    if max(abs(x - x_old)) < tol
        break;
    end

end

T = x + 25;


fprintf('Aantal iteraties: %d\n', k);

fprintf('\nTemperatuurstijgingen:\n');
fprintf('theta1 = %.6f\n', x(1));
fprintf('theta2 = %.6f\n', x(2));
fprintf('theta3 = %.6f\n', x(3));
fprintf('theta4 = %.6f\n', x(4));

fprintf('\nWerkelijke temperaturen:\n');
fprintf('T1 = %.6f C\n', T(1));
fprintf('T2 = %.6f C\n', T(2));
fprintf('T3 = %.6f C\n', T(3));
fprintf('T4 = %.6f C\n', T(4));
